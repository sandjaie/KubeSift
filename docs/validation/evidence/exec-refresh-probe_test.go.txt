package authprobe

import (
 "fmt"
 "net/http"
 "net/http/httptest"
 "os"
 "path/filepath"
 "strings"
 "sync/atomic"
 "testing"

 "k8s.io/client-go/rest"
 api "k8s.io/client-go/tools/clientcmd/api"
)

type guard struct { next http.RoundTripper; approved *atomic.Bool; checks *atomic.Int32 }
func(g guard)RoundTrip(req *http.Request)(*http.Response,error){g.checks.Add(1);if !g.approved.Load(){return nil,fmt.Errorf("trust_required")};return g.next.RoundTrip(req)}

func TestRevocationBeforeNative401Refresh(t *testing.T){
 dir:=t.TempDir();counter:=filepath.Join(dir,"calls");helper:=filepath.Join(dir,"helper")
 script:="#!/bin/sh\necho invocation >> \"$1\"\nprintf '%s' '{\"apiVersion\":\"client.authentication.k8s.io/v1\",\"kind\":\"ExecCredential\",\"status\":{\"token\":\"synthetic-fixture\"}}'\n"
 if err:=os.WriteFile(helper,[]byte(script),0700);err!=nil{t.Fatal(err)}
 var approved atomic.Bool;approved.Store(true);var checks atomic.Int32
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){approved.Store(false);w.WriteHeader(http.StatusUnauthorized)}));defer server.Close()
 cfg:=&rest.Config{Host:server.URL,ExecProvider:&api.ExecConfig{Command:helper,Args:[]string{counter},APIVersion:"client.authentication.k8s.io/v1",InteractiveMode:api.NeverExecInteractiveMode}}
 native,err:=rest.TransportFor(cfg);if err!=nil{t.Fatal(err)}
 client:=&http.Client{Transport:guard{native,&approved,&checks}}
 response,err:=client.Get(server.URL);if err!=nil{t.Fatal(err)};response.Body.Close()
 data,err:=os.ReadFile(counter);if err!=nil{t.Fatal(err)}
 calls:=len(strings.Fields(string(data)))
 t.Logf("request guard checks=%d, helper invocations=%d, trust revoked=%v",checks.Load(),calls,!approved.Load())
 if calls!=1{t.Fatalf("helper executed after revocation: got %d invocations, want 1",calls)}
}
